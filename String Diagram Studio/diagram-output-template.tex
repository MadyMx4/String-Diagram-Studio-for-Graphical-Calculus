\newcommand{\dMult}[6][center]{%
  \begingroup
  \def\dMultTargetOffset{#1}%
  \def\dMultCenter{center}%
  \draw[ultra thick, line cap=round, line join=round, color=#6]
    (#2,#3) -- (#2,#3 + #5 * 0.2)
    .. controls (#2,#3 + #5 * 0.5) .. (#2 + #4 * 0.25,#3 + #5 * 0.5)
    -- (#2 + #4 * 0.5,#3 + #5 * 0.5)
    -- (#2 + #4 * 0.75,#3 + #5 * 0.5)
    .. controls (#2 + #4,#3 + #5 * 0.5) .. (#2 + #4,#3 + #5 * 0.2)
    -- (#2 + #4,#3);
  \ifx\dMultTargetOffset\dMultCenter
    \draw[ultra thick, line cap=round, line join=round, color=#6]
      (#2 + #4 * 0.5,#3 + #5 * 0.5) -- (#2 + #4 * 0.5,#3 + #5);
  \else
    \draw[ultra thick, line cap=round, line join=round, color=#6]
      (#2 + #1,#3 + #5 * 0.5) -- (#2 + #1,#3 + #5);
  \fi
  \endgroup
}

\newcommand{\dPairing}[6]{%
  \draw[ultra thick, color=#5]
    (#1,#2) -- (#1,#2 + #4 - 1)
    .. controls (#1,#2 + #4 - 0.5) .. (#1 + #3 * 0.48,#2 + #4 - 0.5)
    -- (#1 + #3 * 0.52,#2 + #4 - 0.5)
    .. controls (#1 + #3,#2 + #4 - 0.5) .. (#1 + #3,#2 + #4 - 1)
    -- (#1 + #3,#2);
  \if\relax\detokenize{#6}\relax
  \else
    \node at (#1 + #3 * 0.5,#2 + #4 - 0.5) {#6};
  \fi
}

\newcommand{\dCopairing}[6]{%
  \draw[ultra thick, color=#5]
    (#1,#2) -- (#1,#2 - #4 + 1)
    .. controls (#1,#2 - #4 + 0.5) .. (#1 + #3 * 0.48,#2 - #4 + 0.5)
    -- (#1 + #3 * 0.52,#2 - #4 + 0.5)
    .. controls (#1 + #3,#2 - #4 + 0.5) .. (#1 + #3,#2 - #4 + 1)
    -- (#1 + #3,#2);
  \if\relax\detokenize{#6}\relax
  \else
    \node at (#1 + #3 * 0.5,#2 - #4 + 0.5) {#6};
  \fi
}
